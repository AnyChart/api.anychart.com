 * Setter for the link fill.

 * <ul>

 *   <li><b>Shortcut:</b> sets the fill of the normal state only; equivalent to {@link anychart.core.dependencyWheel.elements.Link#normal}().fill(value).</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * @example anychart.core.dependencyWheel.elements.Link.fill_set
 * @param {(anychart.graphics.vector.Fill|FillFunction)=} opt_fill [returns the source node color at 65% opacity] Fill to set.
 * @return {anychart.core.dependencyWheel.elements.Link} Self instance for method chaining.

/**
 * Getter for the link hatch fill.
 * @shortDescription Link hatch fill.
 * @category Interactivity
 * @example anychart.core.dependencyWheel.elements.Link.hatchFill_get
 * @return {(anychart.graphics.vector.HatchFill|Function|boolean|null|undefined)} The hatch fill in the normal state.
 * @since 9.0.0
 */
anychart.core.dependencyWheel.elements.Link.prototype.hatchFill;

/**
 * Setter for the link hatch fill.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> not set; the getter returns undefined and no hatch pattern is drawn over the links.</li>
 *   <li><b>Accepted values:</b> a hatch type name such as 'grid' or 'diagonal-cross', optionally followed by color, thickness and size; an object with <b>type</b>, <b>color</b>, <b>thickness</b> and <b>size</b> fields; or a function returning a hatch fill.</li>
 *   <li><b>Omitted fields:</b> color, thickness and size default to 'black 0.5', 1 and 10.</li>
 *   <li><b>true:</b> takes the pattern from the chart hatch fill palette; links take the palette items in turn by their index.</li>
 *   <li><b>null/false:</b> clears the hatch fill; the getter returns null and the pattern is removed.</li>
 *   <li><b>Invalid values:</b> an unknown type name or an object without a known <b>type</b> is read back as a 'backward-diagonal' hatch fill.</li>
 *   <li><b>Shortcut:</b> sets the hatch fill of the normal state only; equivalent to normal().hatchFill(value).</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.DependencyWheel#hatchFillPalette} for the patterns true takes.
 * @example anychart.core.dependencyWheel.elements.Link.hatchFill_set
 * @param {(anychart.graphics.vector.HatchFill.HatchFillType|string|Object|Function|boolean|null)=} opt_type [undefined] Hatch type name, settings object, function, or a boolean to turn hatching on or off.
 * @param {string=} opt_color ['black 0.5'] Hatch color.
 * @param {number=} opt_thickness [1] Hatch line thickness.
 * @param {number=} opt_size [10] Hatch pattern size.
 * @return {anychart.core.dependencyWheel.elements.Link} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dependencyWheel.elements.Link.prototype.hatchFill;
