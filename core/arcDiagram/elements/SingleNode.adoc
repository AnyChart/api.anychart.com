/**
 * Single node settings, returned by {@link anychart.charts.ArcDiagram#node} when called with a node name.<br/>
 * Created lazily the first time a given node id is addressed and lives for the lifetime of the chart. Its own
 * values override the all-nodes {@link anychart.core.arcDiagram.elements.Node} settings for that one node only;
 * anything left unset here falls through to the all-nodes settings, and from there to the chart's automatic default.
 * @param {anychart.charts.ArcDiagram} chart Chart instance.
 * @param {string} id Node id.
 * @constructor
 * @extends {anychart.core.Base}
 * @since 9.0.0
 */
anychart.core.arcDiagram.elements.SingleNode;

/**
 * Getter for this node's own fill.
 * @shortDescription This node's own fill.
 * @category Coloring
 * @example anychart.core.arcDiagram.elements.SingleNode.fill_get
 * @return {(anychart.graphics.vector.Fill|FillFunction)} The fill in the normal state.
 * @since 9.0.0
 */
anychart.core.arcDiagram.elements.SingleNode.prototype.fill;

/**
 * Setter for this node's own fill.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> undefined, so this node falls through to {@link anychart.core.arcDiagram.elements.Node#fill}, and from there to the automatic palette color.</li>
 *   <li><b>Accepted values:</b> the same values {@link anychart.graphics.vector.Fill} accepts (a color string, an array or object of gradient/pattern settings), or a function returning a fill.</li>
 *   <li><b>Shortcut:</b> sets the fill of the normal state only; equivalent to {@link anychart.core.arcDiagram.elements.SingleNode#normal}().fill(value).</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.ArcDiagram#node} for how to get a single node by name.
 * @example anychart.core.arcDiagram.elements.SingleNode.fill_set
 * @param {(anychart.graphics.vector.Fill|FillFunction)=} opt_fill Fill to set.
 * @return {anychart.core.arcDiagram.elements.SingleNode} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.arcDiagram.elements.SingleNode.prototype.fill;
