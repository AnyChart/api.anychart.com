/**
 * Waffle chart cell settings object.<br/>
 * Holds the grid geometry (padding, aspect ratio, size) and, per state, the appearance of the cells
 * (fill, stroke, hatch fill, shape, corner radius). Get it with {@link anychart.charts.Waffle#cell}.
 * @constructor
 * @extends {anychart.core.Base}
 * @since 9.0.0
 */
anychart.core.waffle.Cell;

/**
 * Getter for the cell aspect ratio.
 * @shortDescription Cell aspect ratio.
 * @category Size and Position
 * @example anychart.core.waffle.Cell.aspectRatio_get
 * @return {number} The cell aspect ratio.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.aspectRatio;

/**
 * Setter for the cell aspect ratio.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 1 (square cells).</li>
 *   <li><b>Meaning:</b> the cell's width-to-height ratio; the cell width is {@link anychart.core.waffle.Cell#size} multiplied by this ratio, so it only affects a fixed-size grid.</li>
 * </ul>
 * @example anychart.core.waffle.Cell.aspectRatio_set
 * @param {number=} opt_ratio [1] Cell aspect ratio to set.
 * @return {anychart.core.waffle.Cell} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.aspectRatio;

 */
anychart.core.waffle.Cell.prototype.cornerRadius;

/**
 * Setter for the cell corner radius.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 0.</li>
 *   <li><b>Applies to:</b> square cells only ({@link anychart.core.waffle.Cell#shape} 'square'); every other shape ignores it.</li>
 *   <li><b>Shortcut:</b> cornerRadius() is a shortcut to normal().cornerRadius(); it also accepts a function, called per cell with <code>this.cellIndex</code>, <code>this.row</code>, <code>this.column</code> and <code>this.state</code>.</li>
 * </ul>
 * @example anychart.core.waffle.Cell.cornerRadius_set
 * @param {(number|Function)=} opt_radius [0] Corner radius in pixels to set, or a function returning it per cell.
 * @return {anychart.core.waffle.Cell} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.cornerRadius;

anychart.core.waffle.Cell.prototype.fill;

/**
 * Setter for the cell fill.
 * @detailed
 * <ul>
 *   <li><b>Shortcut:</b> fill() is a shortcut to normal().fill(); accepts every form {@link anychart.core.StateSettings#fill} does (color string, object, function, gradient).</li>

 *   <li><b>No theme default:</b> a freshly drawn chart reads cell().fill() back as undefined; the visible color comes from the chart-level {@link anychart.charts.Waffle#normal} fill until this is set.</li>
 * </ul>
 * See {@link anychart.core.StateSettings#fill} for the full set of accepted forms.
 * @example anychart.core.waffle.Cell.fill_set
 * @param {(anychart.graphics.vector.Fill|FillFunction)=} opt_fill Fill settings to set.
 * @return {anychart.core.waffle.Cell} Self instance for method chaining.
 * @since 9.0.0
 */

/**
 * Getter for the cell shape.
 * @shortDescription Cell shape.
 * @category Specific settings
 * @example anychart.core.waffle.Cell.shape_get
 * @return {anychart.enums.WaffleCellShape|string} The cell shape.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.shape;

/**
 * Setter for the cell shape.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 'square'.</li>
 *   <li><b>Accepted values:</b> the {@link anychart.enums.WaffleCellShape} members - every marker shape except the two line shapes.</li>
 *   <li><b>Case:</b> matched without regard to case.</li>
 *   <li><b>Invalid input:</b> an unrecognized value does not throw, it silently falls back to 'square', not to the previously set shape.</li>
 *   <li><b>Shortcut:</b> shape() is a shortcut to normal().shape(); it also accepts a function, called per cell with <code>this.cellIndex</code>, <code>this.row</code>, <code>this.column</code> and <code>this.state</code>. Use {@link anychart.core.waffle.Cell#cornerRadius} to round the corners of square cells.</li>
 * </ul>
 * @example anychart.core.waffle.Cell.shape_set
 * @param {(anychart.enums.WaffleCellShape|string|Function)=} opt_shape ['square'] Cell shape to set, or a function returning it per cell.
 * @return {anychart.core.waffle.Cell} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.shape;

/**
 * Getter for the cell size.
 * @shortDescription Cell size.
 * @category Size and Position
 * @example anychart.core.waffle.Cell.size_get
 * @return {number} The cell height in pixels, or 0 when the size is auto-fitted.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.size;

/**
 * Setter for the cell size.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 0, which auto-fits the grid to the chart bounds.</li>
 *   <li><b>Fixed size:</b> any other value is the cell height in pixels; the cell width is this value multiplied by {@link anychart.core.waffle.Cell#aspectRatio}. A fixed-size grid does not shrink to fit - if it outgrows the chart area, it stays centered and its edges are cut off.</li>
 * </ul>
 * @example anychart.core.waffle.Cell.size_set
 * @param {number=} opt_size [0] Cell height in pixels to set. 0 means auto-fit.
 * @return {anychart.core.waffle.Cell} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.waffle.Cell.prototype.size;
