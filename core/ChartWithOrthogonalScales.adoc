/**
 * A base class for charts with orthogonal (X and Y) scales and series, such as cartesian, scatter, radar and polar charts.<br/>
 * Exposes the performance-related settings shared by every series drawn on the chart: how many points are rendered before
 * decimation thins them, which decimation algorithm runs, and when Canvas/WebGL boost rendering takes over.
 * @constructor
 * @extends {anychart.core.SeparateChart}
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.ChartWithOrthogonalScales.prototype.maxPointsRendered
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the maximum number of points rendered per series before decimation thins them.
 * @shortDescription Point rendering cap.
 * @category Specific settings
 * @example anychart.core.ChartWithOrthogonalScales.maxPointsRendered_get
 * @return {number} The point cap.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.maxPointsRendered;

/**
 * Setter for the maximum number of points rendered per series before decimation thins them.<br/>
 * A series above the cap is thinned by decimationAlgorithm() down to this many points.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 1500.</li>
 *   <li><b>Zero:</b> 0 disables the cap, so every point is drawn and decimation never runs.</li>
 *   <li><b>Value type:</b> rounds to the nearest whole number.</li>
 *   <li><b>Invalid values:</b> a non-numeric value is stored as NaN and read back as NaN, whether the previous value was the default or an explicitly set number; it is not cleared back to the default.</li>
 *   <li><b>Skips:</b> decimation is also skipped for stacked series, for size-encoded series such as bubble, for heat map series always - thinning selects points along one ordered axis, and a categorical matrix has no midpoint to collapse neighboring cells into - and whenever boost draws the points instead.</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.Cartesian#maxPointsRendered} for the chart-level setting.
 * @example anychart.core.ChartWithOrthogonalScales.maxPointsRendered_set
 * @param {number=} opt_value [1500] Point cap to set.
 * @return {anychart.core.ChartWithOrthogonalScales} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.maxPointsRendered;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.ChartWithOrthogonalScales.prototype.decimationAlgorithm
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the decimation algorithm used to thin series points above maxPointsRendered().
 * @shortDescription Decimation algorithm.
 * @category Specific settings
 * @example anychart.core.ChartWithOrthogonalScales.decimationAlgorithm_get
 * @return {(anychart.enums.DecimationAlgorithm|string)} The algorithm.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.decimationAlgorithm;

/**
 * Setter for the decimation algorithm used to thin series points above maxPointsRendered().
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 'auto', which picks 'min-max' for discrete series (one mark per point, such as column or marker) and 'lttb' for continuous ones (such as line or area).</li>
 *   <li><b>'lttb':</b> Largest-Triangle-Three-Buckets, keeps the visual shape of a continuous curve.</li>
 *   <li><b>'min-max':</b> keeps the extremes of every bucket, so peaks and troughs survive.</li>
 *   <li><b>Invalid values:</b> any value is coerced to a string and stored verbatim, including a name that is none of the three above; decimation then runs as 'min-max' for it, the same as for any other unrecognized string.</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.Cartesian#decimationAlgorithm} for the chart-level setting.
 * @example anychart.core.ChartWithOrthogonalScales.decimationAlgorithm_set
 * @param {(anychart.enums.DecimationAlgorithm|string)=} opt_value ['auto'] Algorithm to set.
 * @return {anychart.core.ChartWithOrthogonalScales} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.decimationAlgorithm;

/**
 * Getter for the point count above which boost auto-enables for a series.
 * @shortDescription Boost auto-enable threshold.
 * @category Specific settings
 * @example anychart.core.ChartWithOrthogonalScales.boostThreshold_get
 * @return {number} The threshold.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.boostThreshold;

/**
 * Setter for the point count above which boost auto-enables for a series.<br/>
 * Only applies while boostEnabled() is left at its default; an explicit boostEnabled() bypasses it.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> 5000.</li>
 *   <li><b>Zero:</b> 0 never auto-enables boost by point count alone; boost still runs on a series whose boostEnabled() is explicitly true.</li>
 *   <li><b>Value type:</b> rounds to the nearest whole number.</li>
 *   <li><b>Invalid values:</b> a non-numeric value is stored as NaN and read back as NaN; it is not cleared back to the default.</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.Cartesian#boostThreshold} for the chart-level setting.
 * @example anychart.core.ChartWithOrthogonalScales.boostThreshold_set
 * @param {number=} opt_value [5000] Threshold to set.
 * @return {anychart.core.ChartWithOrthogonalScales} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.boostThreshold;

/**
 * Getter for whether the chart's series draw through Canvas/WebGL boost rendering.
 * @shortDescription Boost rendering.
 * @category Specific settings
 * @example anychart.core.ChartWithOrthogonalScales.boostEnabled_get
 * @return {?boolean} The boost state, or null for "auto".
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.boostEnabled;

/**
 * Setter for whether the chart's series draw through Canvas/WebGL boost rendering.
 * @detailed
 * <ul>
 *   <li><b>Default:</b> null, "auto" - a series boosts once its point count passes boostThreshold(), but only while it is the only enabled series on the plot; never applies to a stacked, 3D, OHLC or range-based series, or to a series whose drawer combines line-based and discrete-based rendering.</li>
 *   <li><b>true:</b> forces boost on for every eligible series regardless of boostThreshold(); a series whose drawer has no boost renderer still falls back to normal drawing.</li>
 *   <li><b>false:</b> turns boost off for every series, however many points it has.</li>
 *   <li><b>Invalid values:</b> any value other than null is coerced with Boolean(), so a non-empty string or an object is stored as true and 0 or an empty string is stored as false.</li>
 *   <li><b>Redraw:</b> takes effect on a chart that is already drawn.</li>
 * </ul>
 * See {@link anychart.charts.Cartesian#boostEnabled} for the chart-level setting.
 * @example anychart.core.ChartWithOrthogonalScales.boostEnabled_set
 * @param {?boolean=} opt_value [null] Boost state to set.
 * @return {anychart.core.ChartWithOrthogonalScales} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.ChartWithOrthogonalScales.prototype.boostEnabled;
