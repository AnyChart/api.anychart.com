/**
 * Getter for the bounds mode.
 * @shortDescription Bounds mode settings.
 * @category Interactivity
 * @detailed The bounds mode defines how far the tree can be moved out of the chart area.
 * <ul>
 *   <li><b>Default:</b> 'contain'.</li>
 * </ul>
 * @listing See listing
 * var boundsMode = chart.interactivity().boundsMode();
 * @example anychart.core.orgChart.Interactivity.boundsMode_get
 * @return {(anychart.enums.BoundsMode|string)} The bounds mode.
 * @since 9.0.0
 */
anychart.core.orgChart.Interactivity.prototype.boundsMode;

/**
 * Setter for the bounds mode.
 * @detailed <ul>
 *   <li><b>'free':</b> the tree can be panned without any limit.</li>
 *   <li><b>'contain':</b> the tree is kept inside the chart area on every step of a gesture; when the tree is bigger than the area, its far edge stops at the edge of the area.</li>
 *   <li><b>'auto-center':</b> a gesture is not limited while it runs; when a gesture ends with every node box fully out of the chart area, the tree animates back to the center in about 300 ms, and the zoom level does not change.</li>
 *   <li><b>Ignored:</b> any other value; such a call writes a warning to the console and keeps the current mode.</li>
 * </ul>
 * @listing See listing
 * chart.interactivity().boundsMode('auto-center');
 * @example anychart.core.orgChart.Interactivity.boundsMode_set
 * @param {(anychart.enums.BoundsMode|string)} value ['contain'] The bounds mode.
 * @return {anychart.core.orgChart.Interactivity} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.orgChart.Interactivity.prototype.boundsMode;

/**
 * Getter for the zoom step.
 * @shortDescription Zoom step settings.
 * @category Interactivity
 * @detailed <ul>
 *   <li><b>Default:</b> 1.2.</li>
 * </ul>
 * @listing See listing
 * var zoomStep = chart.interactivity().zoomStep();
 * @example anychart.core.orgChart.Interactivity.zoomStep_get
 * @return {number} The zoom step.
 * @since 9.0.0
 */
anychart.core.orgChart.Interactivity.prototype.zoomStep;

/**
 * Setter for the zoom step.
 * @detailed <ul>
 *   <li><b>Zoom in:</b> {@link anychart.charts.OrgChart#zoomIn} called with no argument, and the zoom-in UI button, multiply the current scale by this step.</li>
 *   <li><b>Zoom out:</b> {@link anychart.charts.OrgChart#zoomOut} called with no argument, and the zoom-out UI button, divide the current scale by this step, except while the step is exactly 1.2, its default value; at exactly 1.2 zoom-out instead uses a fixed 0.8, and any other value switches zoom-out onto the exact 1 / step division.</li>
 *   <li><b>Accepted values:</b> a finite number greater than 1.</li>
 *   <li><b>Invalid input:</b> a value that does not satisfy that (e.g. 1, a negative number, or a non-numeric string) is ignored, and a warning is logged to the console the first time; the current step is always kept.</li>
 * </ul>
 * See {@link anychart.core.orgChart.Interactivity#wheelZoomStep} for the mouse wheel step and {@link anychart.core.orgChart.Interactivity#pinchZoomStep} for the pinch step.
 * @listing See listing
 * chart.interactivity().zoomStep(1.5);
 * @example anychart.core.orgChart.Interactivity.zoomStep_set
 * @param {number} value [1.2] The zoom step to set.
 * @return {anychart.core.orgChart.Interactivity} Self instance for method chaining.
 * @since 9.0.0
 */

 * See {@link anychart.core.orgChart.Interactivity#zoomStep} for the zoomIn()/zoomOut() step and {@link anychart.core.orgChart.Interactivity#pinchZoomStep} for the pinch step.
 * @listing See listing

anychart.core.orgChart.Interactivity.prototype.selectionMode;

/**
 * Setter for the selection mode.<br/>
 * Controls what a click on a card does to the selection; {@link anychart.charts.OrgChart#select} and {@link anychart.charts.OrgChart#unselect} change the selection directly and ignore this setting.
 * @detailed <ul>
 *   <li><b>'none':</b> a click on a card does not select it.</li>
 *   <li><b>'single-select' and 'multi-select':</b> the two modes behave the same way. A plain click on a card replaces the current selection with that card; holding Ctrl, Cmd or Shift while clicking toggles that card in or out of the current selection instead of replacing it.</li>
 *   <li><b>Aliases:</b> 'single' and 'multi' are accepted as short names for 'single-select' and 'multi-select'; every accepted value is matched case-insensitively, with or without the dash.</li>
 *   <li><b>Invalid values:</b> a value other than the ones above is ignored and the mode already in effect is kept, read back unchanged; no console warning is written.</li>
 * </ul>
 * This is the same option {@link anychart.charts.OrgChart#selectionMode} reads and writes.
 * @listing See listing
 * chart.interactivity().selectionMode('single');
 * @example anychart.core.orgChart.Interactivity.selectionMode_set
 * @param {(anychart.enums.SelectionMode|string)} value ['multi-select'] Selection mode to set.
 * @return {anychart.core.orgChart.Interactivity} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.orgChart.Interactivity.prototype.selectionMode;
