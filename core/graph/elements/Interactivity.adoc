/**
 * Getter for the bounds mode.
 * @shortDescription Behavior of the chart content when it is panned out of the chart area.
 * @category Interactivity
 * @since 9.0.0
 * @example anychart.core.graph.elements.Interactivity.boundsMode_get
 * @listing See listing
 * var boundsMode = chart.interactivity().boundsMode();
 * @return {(anychart.enums.BoundsMode|string)} Current bounds mode.
 */
anychart.core.graph.elements.Interactivity.prototype.boundsMode;

/**
 * Setter for the bounds mode.
 * @detailed The bounds mode defines what happens to the graph when it is panned out of the chart area.
 * <ul>
 *   <li><b>Default:</b> 'free'.</li>
 *   <li><b>'free':</b> panning is unbounded, the graph can be dragged anywhere.</li>
 *   <li><b>'contain':</b> the drawn content is clamped to the chart area on every step of a gesture; when the content is bigger than the area, its far edge stops at the edge of the area.</li>
 *   <li><b>'auto-center':</b> nothing is clamped while dragging, but once a gesture ends with every node fully out of the chart area, the content animates back to the center in about 300 ms, and the scale stays unchanged.</li>
 *   <li><b>Invalid values:</b> an unknown value is ignored, and a warning is printed to the console.</li>
 * </ul>
 * @category Interactivity
 * @since 9.0.0
 * @example anychart.core.graph.elements.Interactivity.boundsMode_set
 * @param {(anychart.enums.BoundsMode|string)} value ['free'] Bounds mode.
 * @listing See listing
 * chart.interactivity().boundsMode('contain'); // keep the graph inside the chart area
 * @return {anychart.core.graph.elements.Interactivity} Self instance for method chaining.
 */
anychart.core.graph.elements.Interactivity.prototype.boundsMode;
