/**
 * DataSheet header controller.<br/>
 * Governs the column header row of the grid: whether the header row is rendered at all and how tall it is.<br/>
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#header} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.Header
 * @since 9.0.0
 */
anychart.core.dataSheet.Header;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Header.prototype.enabled
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the header enabled state.
 * @shortDescription Header enabled state.
 * @category Specific settings
 * @example anychart.core.dataSheet.Header.enabled_get
 * @return {boolean} Whether the header row is enabled.
 * @since 9.0.0
 */
anychart.core.dataSheet.Header.prototype.enabled;

/**
 * Setter for the header enabled state.<br/>
 * Set to <b>false</b> to render the grid without a column header row.
 * @example anychart.core.dataSheet.Header.enabled_set
 * @param {boolean=} opt_value [true] Whether to enable the header row.
 * @return {anychart.core.dataSheet.Header} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Header.prototype.enabled;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Header.prototype.height
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the header row height.
 * @shortDescription Header row height.
 * @category Size and Position
 * @example anychart.core.dataSheet.Header.height_get
 * @return {number} Header row height in pixels.
 * @since 9.0.0
 */
anychart.core.dataSheet.Header.prototype.height;

/**
 * Setter for the header row height.<br/>
 * The value is coerced to a number; a non-numeric or zero value falls back to <b>40</b>.
 * @example anychart.core.dataSheet.Header.height_set
 * @param {number=} opt_value [40] Header row height in pixels.
 * @return {anychart.core.dataSheet.Header} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Header.prototype.height;
