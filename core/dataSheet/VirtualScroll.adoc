/**
 * DataSheet virtual scroll controller.<br/>
 * Governs row virtualization: when enabled, only the rows inside the viewport plus a buffer of extra rows above and
 * below it are kept in the DOM, which keeps rendering fast on large datasets. When disabled, every row of the grid is
 * rendered at once.
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#virtualScroll} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.VirtualScroll
 * @since 9.0.0
 */
anychart.core.dataSheet.VirtualScroll;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.VirtualScroll.prototype.enabled
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the virtual scrolling state.
 * @shortDescription Virtual scrolling state.
 * @category Specific settings
 * @listing See listing
 * var virtualScroll = grid.virtualScroll();
 * var state = virtualScroll.enabled();
 * @return {boolean} Whether virtual scrolling is enabled.
 * @since 9.0.0
 */
anychart.core.dataSheet.VirtualScroll.prototype.enabled;

/**
 * Setter for the virtual scrolling state.<br/>
 * When set to <b>false</b> the grid renders all rows at once instead of only the visible range, which may noticeably
 * slow down large datasets.
 * @listing See listing
 * var virtualScroll = grid.virtualScroll();
 * virtualScroll.enabled(false);
 * @param {boolean=} opt_value [true] Whether to enable virtual scrolling.
 * @return {anychart.core.dataSheet.VirtualScroll} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.VirtualScroll.prototype.enabled;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.VirtualScroll.prototype.bufferSize
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the buffer size.<br/>
 * The buffer size is the number of extra rows rendered above and below the viewport.
 * @shortDescription Number of buffer rows above and below the viewport.
 * @category Specific settings
 * @listing See listing
 * var virtualScroll = grid.virtualScroll();
 * var bufferSize = virtualScroll.bufferSize();
 * @return {number} Number of buffer rows rendered above and below the viewport.
 * @since 9.0.0
 */
anychart.core.dataSheet.VirtualScroll.prototype.bufferSize;

/**
 * Setter for the buffer size.<br/>
 * A larger buffer makes fast scrolling smoother at the cost of rendering more rows. The passed value is normalized to a
 * non-negative integer. The buffer is only applied while {@link anychart.core.dataSheet.VirtualScroll#enabled} is
 * <b>true</b>.
 * @listing See listing
 * var virtualScroll = grid.virtualScroll();
 * virtualScroll.bufferSize(25);
 * @param {number=} opt_value [10] Number of buffer rows to render above and below the viewport.
 * @return {anychart.core.dataSheet.VirtualScroll} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.VirtualScroll.prototype.bufferSize;
