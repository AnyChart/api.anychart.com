/**
 * DataSheet Selection controller.<br/>
 * Manages the row selection state of the grid: which data indices are selected, how a row click changes
 * that state, and which selection mode (<b>single</b>, <b>multi</b> or <b>checkbox</b>) is in force.<br/>
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#selection} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.Selection
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.deselect
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Removes a single row from the selection by its data index.<br/>
 * If the index is not currently selected, nothing happens and the grid is not redrawn.
 * @shortDescription Deselect a row.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * selection.deselect(2);
 * @param {number} index Data index of the row to deselect.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.deselect;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.deselectAll
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Clears the whole selection, leaving no selected rows.<br/>
 * If nothing is selected, the call is a no-op and the grid is not redrawn.
 * @shortDescription Deselect all rows.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * selection.deselectAll();
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.deselectAll;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.enabled
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the selection enabled state.
 * @shortDescription Selection enabled state.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * var enabled = selection.enabled();
 * @return {boolean} Whether the selection is enabled.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.enabled;

/**
 * Setter for the selection enabled state.<br/>
 * Disabling the selection also clears all currently selected rows, and {@link anychart.core.dataSheet.Selection#select},
 * {@link anychart.core.dataSheet.Selection#toggle} and {@link anychart.core.dataSheet.Selection#handleClick} stop
 * having any effect while it is off.
 * @listing See listing
 * var selection = grid.selection();
 * selection.enabled(false);
 * @param {boolean=} opt_value [true] Whether to enable the selection.
 * @return {anychart.core.dataSheet.Selection} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.enabled;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.handleClick
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Applies a row click to the selection state, taking the current mode and the Ctrl/Cmd modifier into account.<br/>
 * In <b>single</b> mode the click replaces the selection, and clicking the already selected row clears it.
 * In <b>checkbox</b> mode the click always toggles the row. In <b>multi</b> mode a click with the modifier
 * toggles the row, while a click without it replaces the selection with that single row.<br/>
 * The call does nothing while the selection is disabled.
 * @shortDescription Handle a row click.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * selection.mode('multi');
 * selection.handleClick(3, true);
 * @param {number} index Data index of the clicked row.
 * @param {boolean} ctrlKey Whether the Ctrl/Cmd key was held during the click.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.handleClick;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.isSelected
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Checks whether the row with the given data index is currently selected.
 * @shortDescription Selection state of a row.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * if (selection.isSelected(0)) {
 *   selection.deselect(0);
 * }
 * @param {number} index Data index of the row to check.
 * @return {boolean} Whether the row is selected.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.isSelected;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.mode
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the selection mode.
 * @shortDescription Selection mode.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * var mode = selection.mode();
 * @return {string} Current selection mode.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.mode;

/**
 * Setter for the selection mode.<br/>
 * Accepts <b>single</b>, <b>multi</b> or <b>checkbox</b>; the value is lowercased, and any other value falls back
 * to <b>single</b>. In <b>checkbox</b> mode the grid renders an extra leading checkbox column.<br/>
 * <b>Note:</b> Changing the mode clears the current selection.
 * @listing See listing
 * var selection = grid.selection();
 * selection.mode('checkbox');
 * @param {string=} opt_value ["single"] Selection mode to set.
 * @return {anychart.core.dataSheet.Selection} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.mode;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.select
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Selects the row with the given data index.<br/>
 * In <b>single</b> mode the row replaces the whole selection; in <b>multi</b> and <b>checkbox</b> modes it is added
 * to the already selected rows, and an already selected index is not added twice.<br/>
 * The call does nothing while the selection is disabled.
 * @shortDescription Select a row.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * selection.mode('multi');
 * selection.select(0);
 * selection.select(1);
 * @param {number} index Data index of the row to select.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.select;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.selectedIndices
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the selected data indices.<br/>
 * Returns a copy of the internal array, so modifying the returned array does not change the selection.
 * @shortDescription Selected data indices.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * var indices = selection.selectedIndices();
 * @return {!Array.<number>} Data indices of the selected rows.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.selectedIndices;

/**
 * Setter for the selected data indices.<br/>
 * Replaces the whole selection with a copy of the passed array. The value is applied as is: it is not filtered
 * against the current mode, so passing several indices in <b>single</b> mode selects all of them.
 * @listing See listing
 * var selection = grid.selection();
 * selection.mode('multi');
 * selection.selectedIndices([0, 2, 5]);
 * @param {Array.<number>=} opt_value Data indices of the rows to select.
 * @return {anychart.core.dataSheet.Selection} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.selectedIndices;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Selection.prototype.toggle
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Inverts the selection state of the row with the given data index.<br/>
 * A selected row is deselected; an unselected row is selected - replacing the selection in <b>single</b> mode
 * and joining it in <b>multi</b> and <b>checkbox</b> modes.<br/>
 * The call does nothing while the selection is disabled.
 * @shortDescription Toggle a row.
 * @category Interactivity
 * @listing See listing
 * var selection = grid.selection();
 * selection.mode('checkbox');
 * selection.toggle(1);
 * @param {number} index Data index of the row to toggle.
 * @since 9.0.0
 */
anychart.core.dataSheet.Selection.prototype.toggle;
