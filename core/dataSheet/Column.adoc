/**
 * DataSheet Column controller.<br/>
 * Defines a single grid column: which data field it reads, how the value is typed and formatted, how the column is
 * sized and pinned, and whether its cells can be edited and validated.<br/>
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#column} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.Column
 * @since 9.0.0
 */
anychart.core.dataSheet.Column;

//
//  anychart.core.dataSheet.Column.prototype.field

 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.field;

 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.

 */

/**
 * Getter for the column data type.<br/>
 * Possible values are 'string', 'number', 'date' and 'boolean'. The type drives value comparison, formatting and the
 * cell editor used for the column.
 * @shortDescription Column data type.
 * @category Data
 * @example anychart.core.dataSheet.Column.dataType_get
 * @return {string} Column data type.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.dataType;

/**
 * Setter for the column data type.<br/>
 * The passed value is cast to string and lowercased. When the type is not set explicitly, the grid auto-detects it
 * from the first rows of the data.
 * @example anychart.core.dataSheet.Column.dataType_set
 * @param {string=} opt_value ['string'] Column data type: 'string', 'number', 'date' or 'boolean'.
 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.dataType;

anychart.core.dataSheet.Column.prototype.format;

/**
 * Getter for the column sizing mode.<br/>
 * Possible values are 'fixed' (the column keeps its width), 'auto' (the width is measured from the content) and
 * 'flex' (the column shares free space according to its flex weight).
 * @shortDescription Column sizing mode.
 * @category Size and Position
 * @example anychart.core.dataSheet.Column.sizing_get
 * @return {string} Column sizing mode.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.sizing;

/**
 * Setter for the column sizing mode.<br/>
 * The passed value is cast to string and lowercased; any value other than 'fixed', 'auto' or 'flex' falls back
 * to 'fixed'.
 * @example anychart.core.dataSheet.Column.sizing_set
 * @param {string=} opt_value ['fixed'] Column sizing mode: 'fixed', 'auto' or 'flex'.
 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.sizing;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Column.prototype.flex
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the column flex weight.<br/>
 * The weight is only taken into account when the column sizing mode is 'flex'.
 * @shortDescription Column flex weight.
 * @category Size and Position
 * @example anychart.core.dataSheet.Column.flex_get
 * @return {number} Column flex weight.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.flex;

/**
 * Setter for the column flex weight.<br/>
 * The passed value is cast to number, a zero or non-numeric value falls back to 1, and negative values are clamped
 * to 0.
 * @example anychart.core.dataSheet.Column.flex_set
 * @param {number=} opt_value [1] Column flex weight.
 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.flex;

/**
 * Getter for the column pin state.<br/>
 * A pinned column is frozen at the left or the right edge of the grid and is excluded from drag reordering.
 * @shortDescription Column pin state.
 * @category Size and Position
 * @example anychart.core.dataSheet.Column.pinned_get
 * @return {string|boolean} Column pin state: 'left', 'right' or false.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.pinned;

/**
 * Setter for the column pin state.<br/>
 * Only 'left' and 'right' pin the column; any other value unpins it by setting false.
 * @example anychart.core.dataSheet.Column.pinned_set
 * @param {(string|boolean)=} opt_value [false] Column pin state: 'left', 'right' or false.
 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.pinned;

//
//  anychart.core.dataSheet.Column.prototype.editable
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the column editable state.<br/>
 * Cells are not editable by default, so a column has to be marked editable explicitly to let the cell editor open
 * on it.
 * @shortDescription Column editable state.
 * @category Interactivity
 * @example anychart.core.dataSheet.Column.editable_get
 * @return {boolean} Column editable state.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.editable;

/**
 * Setter for the column editable state.<br/>
 * The passed value is cast to boolean. Editing is off by default, both for the column and for the grid cell editor,
 * so both have to be turned on for cell editing to work.
 * @example anychart.core.dataSheet.Column.editable_set
 * @param {boolean=} opt_value [false] Whether the column cells can be edited.
 * @return {anychart.core.dataSheet.Column} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Column.prototype.editable;
