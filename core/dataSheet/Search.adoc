/**
 * DataSheet Search controller.<br/>
 * Governs the inline search bar: it scans every cell of every visible row for the current query,
 * collects the matches, highlights them in the grid and navigates between them. The bar is also
 * opened by the Ctrl+F (Cmd+F) shortcut, and inside it Enter/Shift+Enter step through matches while
 * Esc closes it.<br/>
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#search} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.Search
 * @since 9.0.0
 */
anychart.core.dataSheet.Search;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.enabled
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the search state.<br/>
 * When search is disabled the bar cannot be shown at all.
 * @shortDescription Search state.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.enabled_get
 * @return {boolean} Whether search is enabled.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.enabled;

/**
 * Setter for the search state.
 * @example anychart.core.dataSheet.Search.enabled_set
 * @param {boolean=} opt_value [true] Whether to enable search.
 * @return {anychart.core.dataSheet.Search} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.enabled;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.query
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the current search query.
 * @shortDescription Search query.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.query_get
 * @return {string} Current search query.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.query;

/**
 * Setter for the search query.<br/>
 * Setting the query runs the search immediately: matching is case-insensitive and is performed as a
 * substring test over the string representation of every cell value. The first match, if any, becomes
 * the current one and the viewport is scrolled to it.
 * @example anychart.core.dataSheet.Search.query_set
 * @param {string=} opt_value Query to search for.
 * @return {anychart.core.dataSheet.Search} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.query;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.matchCount
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Gets the number of cells matching the current query.<br/>
 * Returns 0 when the query is empty or nothing was found.
 * @shortDescription Number of matches.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.matchCount
 * @return {number} Number of matched cells.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.matchCount;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.show
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Shows the search bar.<br/>
 * Does nothing if search is disabled. The input field receives focus once the bar is rendered.
 * @shortDescription Shows the search bar.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.show
 * @return {undefined}
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.show;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.close
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Closes the search bar.<br/>
 * Hides the bar, resets the query to an empty string, drops all collected matches and removes the
 * highlighting from the grid.
 * @shortDescription Closes the search bar.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.close
 * @return {undefined}
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.close;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.next
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Navigates to the next match.<br/>
 * Navigation is cyclic: from the last match it wraps back to the first one. The viewport is scrolled
 * to the new current match. Does nothing when there are no matches.
 * @shortDescription Navigates to the next match.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.next
 * @return {undefined}
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.next;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.prev
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Navigates to the previous match.<br/>
 * Navigation is cyclic: from the first match it wraps back to the last one. The viewport is scrolled
 * to the new current match. Does nothing when there are no matches.
 * @shortDescription Navigates to the previous match.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.prev
 * @return {undefined}
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.prev;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Search.prototype.isVisible
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Checks whether the search bar is currently visible.
 * @shortDescription Search bar visibility.
 * @category Interactivity
 * @example anychart.core.dataSheet.Search.isVisible
 * @return {boolean} Whether the search bar is shown.
 * @since 9.0.0
 */
anychart.core.dataSheet.Search.prototype.isVisible;
