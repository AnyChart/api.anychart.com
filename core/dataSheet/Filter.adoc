/**
 * DataSheet filter controller.<br/>
 * Manages per-column filtering of the grid data. Each filter is stored under a data field name and is described by a
 * plain configuration object of one of three types: text (with 'contains', 'startsWith' or 'exact' matching mode),
 * number (with 'min' and/or 'max' bounds) or boolean. A row is displayed only if it passes every active filter.
 * <b>Note:</b> Use {@link anychart.charts.DataSheet#filter} method to get this controller.<br/>
 * @constructor
 * @name anychart.core.dataSheet.Filter
 * @since 9.0.0
 */
anychart.core.dataSheet.Filter;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Filter.prototype.enabled
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Getter for the filtering enabled state.
 * @shortDescription Filtering enabled state.
 * @category Interactivity
 * @example anychart.core.dataSheet.Filter.enabled_get
 * @return {boolean} Whether filtering is enabled.
 * @since 9.0.0
 */
anychart.core.dataSheet.Filter.prototype.enabled;

/**
 * Setter for the filtering enabled state.<br/>
 * When disabled, all rows pass through regardless of the configured filters — the filter configs themselves are kept
 * and take effect again as soon as filtering is switched back on.
 * @example anychart.core.dataSheet.Filter.enabled_set
 * @param {boolean=} opt_value [true] Whether to enable filtering.
 * @return {anychart.core.dataSheet.Filter} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Filter.prototype.enabled;

//----------------------------------------------------------------------------------------------------------------------
//
//  anychart.core.dataSheet.Filter.prototype.filterBy
//
//----------------------------------------------------------------------------------------------------------------------

/**
 * Sets a filter on the given data field.<br/>
 * The configuration object shape depends on its 'type' key:<br/>
 * <ul>
 *   <li><b>text</b> — {type: 'text', mode: 'contains'|'startsWith'|'exact', value: string}, matching is
 *   case-insensitive and 'contains' is used when the mode is omitted; an empty value matches everything.</li>
 *   <li><b>number</b> — {type: 'number', min: number, max: number}, both bounds are inclusive and optional;
 *   non-numeric values never pass.</li>
 *   <li><b>boolean</b> — {type: 'boolean', value: boolean}, the cell value is compared by truthiness.</li>
 * </ul>
 * Setting a filter on a field that already has one replaces it. Only one filter per field is kept.
 * @shortDescription Sets a filter on a field.
 * @category Data
 * @example anychart.core.dataSheet.Filter.filterBy
 * @param {string} field Data field name.
 * @param {Object} config Filter configuration.
 * @return {anychart.core.dataSheet.Filter} Self instance for method chaining.
 * @since 9.0.0
 */
anychart.core.dataSheet.Filter.prototype.filterBy;
