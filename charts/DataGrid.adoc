/**
 * DataGrid chart class.<br/>
 * A DOM-based interactive data grid with virtual scrolling, column sorting, filtering, cell editing, row selection,
 * grouping/hierarchy and clipboard support.<br/>
 * <b>Note:</b> Use {@link anychart.dataGrid} method to get this chart.<br/>
 * @constructor
 * @extends {anychart.core.VisualBaseWithBounds}
 * @since 9.0.0
 */
anychart.charts.DataGrid;

//

 * Getter for the grid data.
 * @shortDescription Data settings.
 * @category Data

 * @since 9.0.0
 */
anychart.charts.DataGrid.prototype.data;
