anychart.charts.Waffle.prototype.selectionMode;

/**
 * Setter for the selection mode.<br/>
 * Defines how mouse clicks select the categories of the chart.
 * @detailed Supported modes:
 * <ul>
 *   <li><b>'none'</b> - clicks select nothing, the chart is never repainted in the selected state.</li>
 *   <li><b>'single-select'</b> - a click replaces the selection, so at most one category is selected at a time.</li>
 *   <li><b>'multi-select'</b> - the default mode, more than one category can be in the selected state at a time.</li>
 *   <li><b>'drill-down'</b> - accepted, but a waffle chart has no drill-down action and the mode behaves as 'single-select'.</li>
 * </ul>
 * <b>Note</b>: an unrecognized value does not throw, it silently falls back to 'none' and thus disables selection.
 * The camelCase aliases 'singleSelect' and 'multiSelect' are recognized and normalize to the hyphenated forms.<br/>
 * <b>Note</b>: the mode gates mouse gestures only. The select() and unselect() methods keep working in any mode, 'none' included.<br/>
 * <b>Note</b>: this method is a shortcut to the selection mode of the chart interactivity settings, so it returns the
 * interactivity settings object and not the chart. It cannot be used as a link in a chart method chain.
 * See {@link anychart.core.utils.Interactivity#selectionMode} for the same setting on the interactivity object.
 * @example anychart.charts.Waffle.selectionMode_set
 * @param {(anychart.enums.SelectionMode|string)=} opt_mode ['multi-select'] Selection mode to set.
 * @return {anychart.core.utils.Interactivity} Chart interactivity settings.
 * @since 9.0.0
 */
anychart.charts.Waffle.prototype.selectionMode;
