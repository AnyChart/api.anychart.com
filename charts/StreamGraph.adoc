anychart.charts.StreamGraph.prototype.offset;

/**
 * Setter for the stream offset.<br/>
 * The offset defines how the layers are displaced around the baseline.
 * @detailed Supported offsets:
 * <ul>
 *   <li><b>'wiggle'</b> - minimizes wiggle of layer slopes.</li>
 *   <li><b>'silhouette'</b> - centers the stream around the x-axis.</li>
 *   <li><b>'expand'</b> - normalizes each stack to fill [0..1].</li>
 *   <li><b>'zero'</b> - plain stacking from the zero baseline.</li>
 * </ul>
 * @example anychart.charts.StreamGraph.offset_set
 * @param {(anychart.enums.StreamGraphOffset|string)=} opt_offset ['wiggle'] Stream offset to set.
 * @return {anychart.charts.StreamGraph} Self instance for method chaining.
 * @since 9.0.0
 */
